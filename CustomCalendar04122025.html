<!DOCTYPE html>

<html>

<head>

  <title>Workout Calendar</title>

  <link href="https://cdn.jsdelivr.net/npm/fullcalendar@6.1.8/main.min.css" rel="stylesheet"/>

  <style>

    #calendar { max-width: 100%; margin: 10px; }

    .fc-daygrid-day-number { color: #333; }

    .fc-event { background-color: #FFD700; border-color: #DAA520; } /* Gold for workouts */

  </style>

</head>

<body>

  <div id="calendar"></div>

  <script src="https://cdn.jsdelivr.net/npm/fullcalendar@6.1.8/main.min.js"></script>

  <script>

    document.addEventListener('DOMContentLoaded', function() {

      var calendarEl = document.getElementById('calendar');

      var calendar = new FullCalendar.Calendar(calendarEl, {

        initialView: 'dayGridMonth',

        initialDate: '2025-04-01',

        events: function(fetchInfo, successCallback) {

          fetch('YOUR_ADALO_API_URL/workouts') // Replace with your API

            .then(response => response.json())

            .then(data => {

              var events = data.map(item => ({

                title: item.Name,

                start: item.Date,

                end: item.Date

              }));

              successCallback(events);

            });

        },

        eventClick: function(info) {

          window.location.href = '/day-workouts?date=' + info.event.startStr;

        }

      });

      calendar.render();

    });

  </script>

</body>

</html>
